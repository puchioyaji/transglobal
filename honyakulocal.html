rec.onresult = (event) => { // 音声合成（TTS）実行中および終了直後はマイクの拾い直し防止のため無視 if (isSpeaking || Date.now() - lastSpeakingEndTime < 400) return; let interimTranscript = ''; let finalTranscript = ''; for (let i = event.resultIndex; i < event.results.length; ++i) { const transcript = event.results[i][0].transcript; if (event.results[i].isFinal) { finalTranscript += transcript; } else { interimTranscript += transcript; } } // 1. 確定データ（isFinal）がある場合は優先的に即時キュー投入 if (finalTranscript.trim()) { if (speechSilenceTimer) clearTimeout(speechSilenceTimer); pendingInterimText = ""; enqueueRecognizedText(finalTranscript.trim()); } // 2. 未確定データ（interimResults）の処理（話始めや短文の取りこぼし防止） else if (interimTranscript.trim()) { pendingInterimText = interimTranscript.trim(); statusText.textContent = `認識中... ${pendingInterimText}`; if (speechSilenceTimer) clearTimeout(speechSilenceTimer); // 一定文字数（例: 8文字以上）に達したら確定を待たずに即時送信して遅延を回避 if (pendingInterimText.length >= 8) { enqueueRecognizedText(pendingInterimText); pendingInterimText = ""; return; } // 短いフレーズで確定フラグが立たない場合の無音タイムアウト（350ms） speechSilenceTimer = setTimeout(() => { if (pendingInterimText.length > 0) { enqueueRecognizedText(pendingInterimText); pendingInterimText = ""; } }, 350); } };
