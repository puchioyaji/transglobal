rec.onresult = (event) => { if (isSpeaking || Date.now() - lastSpeakingEndTime < 500) return; let interimTranscript = ''; let finalTranscript = ''; for (let i = event.resultIndex; i < event.results.length; ++i) { const transcript = event.results[i][0].transcript; if (event.results[i].isFinal) { finalTranscript += transcript; } else { interimTranscript += transcript; } } // 1. 確定データは即時処理 if (finalTranscript.trim()) { if (speechSilenceTimer) clearTimeout(speechSilenceTimer); pendingInterimText = ""; enqueueRecognizedText(finalTranscript.trim()); } // 2. 未確定データの超高速処理 else if (interimTranscript.trim()) { pendingInterimText = interimTranscript.trim(); statusText.textContent = `認識中... ${pendingInterimText}`; if (speechSilenceTimer) clearTimeout(speechSilenceTimer); // 【改善点1】一定文字数（例: 6文字以上）に達したら無音を待たずに即投入 if (pendingInterimText.length >= 6) { enqueueRecognizedText(pendingInterimText); pendingInterimText = ""; return; } // 【改善点2】無音判定タイマーを 750ms から 350ms へ短縮 speechSilenceTimer = setTimeout(() => { if (pendingInterimText.length > 0) { enqueueRecognizedText(pendingInterimText); pendingInterimText = ""; } }, 350); } };
