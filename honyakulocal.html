function initSpeechRecognitionEngine() { if (!SpeechRecognition) return null; const rec = new SpeechRecognition(); rec.continuous = true; rec.interimResults = true; // ★ interimResultsをtrueにして文脈補正を効かせる rec.lang = 'ja-JP'; rec.onresult = (event) => { let interimTranscript = ''; let finalTranscript = ''; for (let i = event.resultIndex; i < event.results.length; ++i) { if (event.results[i].isFinal) { finalTranscript += event.results[i][0].transcript; } else { interimTranscript += event.results[i][0].transcript; } } // 途中の仮認識結果をステータス表示（UIの応答性を維持） if (interimTranscript) { statusText.textContent = `認識中: ${interimTranscript}`; } // 完全に文章として確定（isFinal）したものだけをキューへ送る if (finalTranscript.trim()) { enqueueRecognizedText(finalTranscript.trim()); } }; rec.onend = () => { if (isRecording) { setTimeout(() => { try { recognition.start(); } catch(e){} }, 50); } }; return rec; }
